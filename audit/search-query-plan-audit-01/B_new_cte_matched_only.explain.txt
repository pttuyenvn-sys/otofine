WITH 
    matched AS (
      SELECT DISTINCT
        `p`.`id` AS product_id,
        COALESCE(NULLIF(pc.canonical_name, ''), pc.category_name) AS canonical_name,
        COALESCE(NULLIF(pc.canonical_slug, ''), pc.category_slug) AS canonical_slug,
        cm.hang_xe AS brand,
        cm.ten_xe AS model,
        COALESCE(pc.search_priority, 0) AS search_priority
      
    FROM products p

    LEFT JOIN product_category_map pcm
      ON pcm.product_id = `p`.`id`

    LEFT JOIN product_categories pc
      ON pc.id = pcm.category_id

    LEFT JOIN product_car_applications pa
      ON pa.productId = `p`.`id`

    LEFT JOIN car_models cm
      ON cm.id = pa.carModelId

    JOIN shops s
      ON `s`.id = `p`.`shopId`

    LEFT JOIN address a
      ON a.id = s.provinceId
  
       WHERE 1=1  AND LOWER(TRIM(cm.hang_xe)) = ? 
          AND (
            LOWER(`p`.`partName`) LIKE ?
            OR LOWER(`p`.`shortDescription`) LIKE ?
            OR LOWER(`p`.`description`) LIKE ?
            OR LOWER(`p`.`partNumber`) LIKE ?
          )
           AND TRIM(LOWER(`s`.`public_status`)) = 'public' AND TRIM(LOWER(`p`.`moderation_status`)) = 'approved' AND `p`.`stock` > 0 
    )
   SELECT COUNT(*) c FROM matched
